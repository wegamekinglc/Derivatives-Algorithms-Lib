cmake_minimum_required(VERSION 3.21)
project(DalMarketIvLocalVolReference LANGUAGES CXX)

set(DAL_BUILD_PUBLIC OFF CACHE BOOL "Core-only qualification" FORCE)
set(DAL_BUILD_PYTHON OFF CACHE BOOL "Core-only qualification" FORCE)
set(DAL_BUILD_EXCEL OFF CACHE BOOL "Core-only qualification" FORCE)
set(DAL_BUILD_EXCEL_PORTABLE_TESTS OFF CACHE BOOL "Core-only qualification" FORCE)
set(DAL_PUBLIC_BUILD_TESTS OFF CACHE BOOL "Core-only qualification" FORCE)
set(DAL_CPP_BUILD_EXAMPLES OFF CACHE BOOL "No example changes" FORCE)
set(DAL_CPP_BUILD_BENCHMARKS OFF CACHE BOOL "No production benchmark changes" FORCE)
set(DAL_ENABLE_NATIVE_ARCH OFF CACHE BOOL "Portable qualification" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "Static qualification" FORCE)
add_subdirectory(../.. workspace EXCLUDE_FROM_ALL)
add_executable(dal_market_iv_capture capture.cpp)
target_link_libraries(dal_market_iv_capture PRIVATE DAL::cpp)
target_include_directories(dal_market_iv_capture PRIVATE ../../dal-cpp/externals/rapidjson/include)

enable_testing()
include(GoogleTest)
add_executable(dal_market_iv_a1_mc a1_mc.cpp)
target_link_libraries(dal_market_iv_a1_mc PRIVATE DAL::cpp)
target_include_directories(dal_market_iv_a1_mc PRIVATE ../../dal-cpp/externals/rapidjson/include)
add_executable(dal_market_iv_a1_protocol_tests test_a1_protocol.cpp)
target_link_libraries(dal_market_iv_a1_protocol_tests PRIVATE DAL::cpp gtest_main)
target_include_directories(dal_market_iv_a1_protocol_tests PRIVATE ../../dal-cpp/externals/rapidjson/include)
set(DAL_A1_REQUEST "${CMAKE_CURRENT_BINARY_DIR}/request.json" CACHE FILEPATH "Generated canonical request for protocol tests")
target_compile_definitions(dal_market_iv_a1_protocol_tests PRIVATE DAL_A1_REQUEST="${DAL_A1_REQUEST}")
gtest_discover_tests(dal_market_iv_a1_protocol_tests)
add_executable(dal_market_iv_a1_probe a1_probe.cpp)
target_link_libraries(dal_market_iv_a1_probe PRIVATE DAL::cpp)

add_executable(dal_market_iv_refit_pair refit_pair.cpp)
target_link_libraries(dal_market_iv_refit_pair PRIVATE DAL::cpp)
add_test(NAME market_iv_refit_pair COMMAND dal_market_iv_refit_pair)

add_executable(dal_market_iv_refit_pv test_refit_pv.cpp)
target_link_libraries(dal_market_iv_refit_pv PRIVATE DAL::cpp gtest_main)
gtest_discover_tests(dal_market_iv_refit_pv)
