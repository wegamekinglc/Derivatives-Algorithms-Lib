cmake_minimum_required(VERSION 3.21)
project(passive_sink_study LANGUAGES CXX)

set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(DAL_BUILD_PUBLIC OFF CACHE BOOL "" FORCE)
set(DAL_BUILD_PYTHON OFF CACHE BOOL "" FORCE)
set(DAL_BUILD_EXCEL OFF CACHE BOOL "" FORCE)
set(DAL_BUILD_EXCEL_PORTABLE_TESTS OFF CACHE BOOL "" FORCE)
set(DAL_CPP_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(DAL_CPP_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(DAL_CPP_BUILD_BENCHMARKS ON CACHE BOOL "" FORCE)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../.." dal)

set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
set(INSTALL_GMOCK OFF CACHE BOOL "" FORCE)
add_subdirectory("../../externals/googletest" googletest EXCLUDE_FROM_ALL)
add_executable(passive_sink_tests test_sink.cpp)
target_link_libraries(passive_sink_tests PRIVATE DAL::cpp gtest_main)
dal_apply_platform_options(passive_sink_tests)
add_executable(passive_sink_study sinkstudy.cpp)
target_link_libraries(passive_sink_study PRIVATE DAL::cpp)
dal_apply_platform_options(passive_sink_study)
enable_testing()
add_test(NAME passive_sink_contract COMMAND passive_sink_tests)
