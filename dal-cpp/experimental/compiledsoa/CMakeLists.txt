cmake_minimum_required(VERSION 3.21)
project(compiled_soa_study LANGUAGES CXX)

set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(DAL_BUILD_PUBLIC OFF CACHE BOOL "" FORCE)
set(DAL_BUILD_PYTHON OFF CACHE BOOL "" FORCE)
set(DAL_BUILD_EXCEL OFF CACHE BOOL "" FORCE)
set(DAL_BUILD_EXCEL_PORTABLE_TESTS OFF CACHE BOOL "" FORCE)
set(DAL_CPP_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(DAL_CPP_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(DAL_CPP_BUILD_BENCHMARKS ON CACHE BOOL "" FORCE)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../.." dal)

add_executable(compiled_soa_study soastudy.cpp)
target_link_libraries(compiled_soa_study PRIVATE DAL::cpp)
dal_apply_platform_options(compiled_soa_study)

set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
set(INSTALL_GMOCK OFF CACHE BOOL "" FORCE)
add_subdirectory("../../externals/googletest" googletest EXCLUDE_FROM_ALL)
add_executable(compiled_soa_tests test_soa.cpp)
target_link_libraries(compiled_soa_tests PRIVATE DAL::cpp gtest)
dal_apply_platform_options(compiled_soa_tests)
enable_testing()
add_test(NAME compiled_soa_tests COMMAND compiled_soa_tests)

if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    add_executable(compiled_soa_sanitized_tests EXCLUDE_FROM_ALL test_soa.cpp)
    target_link_libraries(compiled_soa_sanitized_tests PRIVATE DAL::cpp gtest)
    dal_apply_platform_options(compiled_soa_sanitized_tests)
    target_compile_options(compiled_soa_sanitized_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(compiled_soa_sanitized_tests PRIVATE -fsanitize=address,undefined)
endif()
