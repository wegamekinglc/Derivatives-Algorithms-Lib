cmake_minimum_required(VERSION 3.21)
project(Dal502Qualification LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
get_filename_component(DAL_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
if(NOT DEFINED DAL_CORE_BUILD)
    message(FATAL_ERROR "DAL_CORE_BUILD must identify an independently configured core build")
endif()
find_package(Threads REQUIRED)
add_library(qualification_core STATIC IMPORTED)
set_target_properties(qualification_core PROPERTIES
    IMPORTED_LOCATION "${DAL_CORE_BUILD}/dal-cpp/libdal_cpp.a"
    INTERFACE_INCLUDE_DIRECTORIES "${DAL_ROOT}/dal-cpp;${DAL_ROOT}/dal-cpp/externals/rapidjson/include")
add_executable(dal502_driver driver.cpp)
target_link_libraries(dal502_driver PRIVATE qualification_core Threads::Threads)
add_executable(dal502_bbs bbs_driver.cpp)
target_include_directories(dal502_bbs PRIVATE "${DAL_ROOT}/dal-cpp/externals/rapidjson/include")
set(BBS_GTEST "${DAL_ROOT}/dal-cpp/externals/googletest/googletest")
add_executable(dal502_bbs_tests test_bbs.cpp "${BBS_GTEST}/src/gtest-all.cc")
target_include_directories(dal502_bbs_tests PRIVATE
    "${BBS_GTEST}" "${BBS_GTEST}/include" "${DAL_ROOT}/dal-cpp/externals/rapidjson/include")
target_compile_definitions(dal502_bbs_tests PRIVATE DAL502_BBS_CASES="${CMAKE_CURRENT_LIST_DIR}/bbs_scalar_cases.json")
target_link_libraries(dal502_bbs_tests PRIVATE Threads::Threads)
enable_testing()
add_test(NAME dal502_bbs_scalar COMMAND dal502_bbs_tests)
if(EXISTS "${CMAKE_BINARY_DIR}/observed.cpp")
    add_executable(dal502_observed driver.cpp "${CMAKE_BINARY_DIR}/observed.cpp")
    target_link_libraries(dal502_observed PRIVATE qualification_core Threads::Threads)
endif()
